3. Matrix for each alteration type and combine in binary list

Prepare a list of binary matrices. Each matrix is named like the genomic alteration rows = gene in gene list, columns = patients/samples, value = 1 -> mutation is present

#Load processed data file and select columns of interest
snv <- read_xlsx("/Users/lasmey/Desktop/ccRCC/Metadaten_RCC/04_ccRCC_IHC_Mut/PBRM1_BAP1_SETD2_mutations_ccRCC_panel.xlsx") 
snv <- snv %>% select(Tumor, Gene) %>% mutate(mut_level = str_split(snv$SnpEff_Annotation,"\\|",simplify = T)[,2]) 
snv <- snv %>% mutate(mut_level = str_split(snv$mut_level,"&",simplify = T)[,1])

mut_levels_for_matrix = snv %>% pull(mut_level) %>% unique()
gene_list = snv %>% pull(Gene) %>% unique()

#Placeholders 
mut_list_new = list()
placeholder_gene = gene_list[1]
pids = unique(snv$Tumor)
placeholder_pid = pids[1]

for(mut in mut_levels_for_matrix){
  
  #select all SNVs/Indels of the mut_level
  x = snv %>%
    dplyr::filter(mut_level==mut,
                  Gene %in% gene_list | 
                    grepl(paste0(paste0("^", gene_list, "[,;]"), collapse="|"), Gene) |       #allow for multi-gene mutations
                    grepl(paste0(paste0("[,;]", gene_list, "$"), collapse="|"), Gene) ) %>%  
    dplyr::select(Tumor, Gene) %>%
    dplyr::mutate(value=1) %>%
    dplyr::distinct()
  
  
  if(dim(x)[1]!=0){
    x = x %>%
      tidyr::spread(Gene, value)
    
    #transpose
    y = dplyr::select(x, -Tumor) 
    
    rownames(y) = x$Tumor
    y = t(y)
    
    #replace NA by 0
    y[is.na(y)] = 0
  }else{
    y = tibble(0)
    colnames(y) = placeholder_pid
    row.names(y) = placeholder_gene
  }
  
  mut_list_new[[mut]] = as.matrix(y)
}

#Unify matrices
mut_list2 = unify_mat_list(mut_list_new)

4. Print Oncoprint

# keep the order of annotation and mutation list
ord= anno$Tumor

# show every gene and patient
# blank for patients  and.or genes missing 
for (i in seq_along(mut_list2)){
  dif_gen= setdiff(gene_list,rownames(mut_list2[[i]]))
  dif_pat= setdiff(pids,colnames(mut_list2[[i]]))
  
  blank= matrix(ncol=length(dif_pat),nrow=length(rownames(mut_list2[[i]])),0)
  colnames(blank)=dif_pat
  rownames(blank)=rownames(mut_list2[[i]])
  mut_list2[[i]]=cbind(mut_list2[[i]],blank)
  
  blank= matrix(ncol=ncol(mut_list2[[i]]),nrow=length(dif_gen),0)
  colnames(blank)=colnames(mut_list2[[i]])
  rownames(blank)=dif_gen
  mut_list2[[i]]=rbind(mut_list2[[i]],blank)
  mut_list2[[i]]=mut_list2[[i]][,ord]
}
