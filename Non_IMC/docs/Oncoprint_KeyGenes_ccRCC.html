library(tidyverse)

#load mutation calling files and prepare PID 
snv <- as.data.frame(read_xlsx("/Users/lasmey/Desktop/RCC/Metadaten/04_Originalfiles_Not_corrected/SeqData/PBRM1_BAP1_SETD2_mutations_ccRCC_panel.xlsx"))

snv$PID <- ifelse(str_length(snv$PID) == 10,paste0(str_split(snv$PID,"\\.",simplify = TRUE)[,1],".0",str_split(snv$PID,"\\.",simplify = TRUE)[,2]),snv$PID)

#load full metadata and select relevant patients and metadata
full_anno = read_csv("/Users/lasmey/Desktop/RCC/Metadaten/Metadata_combined_RCC_onlyPT.csv", col_names = TRUE) %>% filter(sequencing_data == TRUE)
full_anno$PID_short <- str_split(full_anno$PID, "-",simplify = TRUE)[,1]

anno <- full_anno %>% 
  filter(PID_short %in% snv$PID) %>%
  select(PID_short,gender, age_bin, subtype, ISUP, pT_simple, necrose, lymphocytes, survival_data) %>%
  mutate_all(as.character)

anno[is.na(anno)] <- "Unknown"

#Load RCC color assignment (Set in Metadata_Update.Rmd)
col_list_1 <- readRDS(file="/Users/lasmey/Desktop/RCC/Metadaten/color.rds")

ha <- HeatmapAnnotation(gender = anno$gender,
