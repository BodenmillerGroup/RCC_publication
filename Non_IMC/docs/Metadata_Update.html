Randomized Stratified Sampling based on clinical metadata for targeted sequencing

###Filter for major RCC classes and cases for which we do not have any sequencing data but do have tissue on the TMA
Meta_seq = Meta_PT %>% 
  filter(subtype %in% c("clear_cell_RCC","papillary_RCC","chromophobe_RCC") & sequencing_data == "FALSE" & no_tissue == "FALSE" & survival_data == "TRUE" & lymphocytes %in% c("1","2")) %>%
  mutate(pT_comb = ifelse(pT_simple == 1|pT_simple == 2, "1/2", 
                  ifelse(pT_simple == 3, "3", 
                         ifelse(pT_simple == 4, "4", NA)))) %>% 
  filter(!is.na(pT_comb)) %>% 
  select(PID, gender, age_bin, subtype, ISUP,pT_simple, "pT12/34",pT_comb, necrose, lymphocytes, BAP1_IHC, PBRM1_IHC, H3K36me_IHC, survival_data,sequencing_data)
