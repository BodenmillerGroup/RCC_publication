Meta_2 <- Meta_1

                                   ifelse(40<age & age<=55, "41-55",

Print heatmap based overview

based on the ComplexHeatmap package and the implemented HeatmapAnnotation function

#Convert NAs for easier handling with ComplexHeatmap package
Meta_2[is.na(Meta_2)] <- "Unknown"

Meta_2 = Meta_2 %>% arrange(desc(type),subtype, desc(ISUP_grade))

#Color scheme
#colnames(Meta_2)
col = list(PID = colorRampPalette(brewer.pal(9, "Set1"))(length(unique(Meta_2$PID))),
           gender = c("#7FC97F", "#BEAED4", "black"),
           age_bin = c(colorRampPalette(brewer.pal(9, "Greens"))(length(unique(Meta_2$age_bin))-1),"black"),
           type = c("#E41A1C", "#377EB8"),
           subtype = c(colorRampPalette(brewer.pal(12, "Paired"))(length(unique(Meta_2$subtype))-1),"black"),
           pT_simple = c(colorRampPalette(brewer.pal(9, "Blues"))(length(unique(Meta_2$pT_simple))-1),"black"),
           ISUP_grade = c(colorRampPalette(brewer.pal(9, "Reds"))(length(unique(Meta_2$ISUP_grade))-1),"black"),
           necrose = c(colorRampPalette(brewer.pal(9, "Purples"))(length(unique(Meta_2$necrose))-1),"black"),
           lymphocytes = c(colorRampPalette(brewer.pal(9, "Greens"))(length(unique(Meta_2$lymphocytes))-1),"black"),
           survival_data = c("black","grey")
)

col_list_1 = foreach(i=1:10, .packages="foreach") %do% {
  color= col[[i]]
  names(color)= sort(unique(Meta_2[,names(col)[i]]))
  color
}

names(col_list_1) <- c(names(col))

#Heatmap Annotation
sample_anno = HeatmapAnnotation(PID = Meta_2$PID,
                                gender = Meta_2$gender,
                                age_bin = Meta_2$age_bin,
                                type = Meta_2$type,
                                subtype = Meta_2$subtype,
                                ISUP_grade = Meta_2$ISUP_grade,
                                pT_simple = Meta_2$pT_simple,
                                necrose = Meta_2$necrose,
                                lymphocytes = Meta_2$lymphocytes,
                                survival_data = Meta_2$survival_data,
                                col = list(PID = col_list_1[["PID"]],
                                           gender = col_list_1[["gender"]],
                                           age_bin = col_list_1[["age_bin"]],
                                           type = col_list_1[["type"]],
                                           subtype = col_list_1[["subtype"]],
                                           ISUP_grade = col_list_1[["ISUP_grade"]],
                                           pT_simple = col_list_1[["pT_simple"]],
                                           necrose = col_list_1[["necrose"]],
                                           lymphocytes = col_list_1[["lymphocytes"]],
                                           survival_data = col_list_1[["survival_data"]]
                                ),
                                border = T,
                                na_col = "black",
                                gap = unit(5, "points"),
                                show_annotation_name = T,
                                show_legend = c(PID = FALSE)
                                #annotation_label = c("PID (n=367)","Gender","Subtype","pT","pN")
)%v%NULL

#Display plot
draw(sample_anno)
