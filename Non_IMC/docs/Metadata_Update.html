Aim: Create updated metasheet for full RCC cohort and visualize a cohort heatmap

Load metadata csv files

library(tidyverse)
library(readxl)
library(knitr)

Meta_A <- as.data.frame(read_xls("/Users/lasmey/Desktop/ccRCC/Metadaten_RCC/TMA224_update_LM.xls",col_names = T))
Meta_B <- as.data.frame(read_xls("/Users/lasmey/Desktop/ccRCC/Metadaten_RCC/ZTMA20_update_LM.xls",col_names = T))

Meta_A <- Meta_A %>% mutate_all(as.character)
Meta_B <- Meta_B %>% mutate_all(as.character)

Meta <- bind_rows(Meta_A,Meta_B)
Meta$PID <- str_replace_all(Meta$PID," ", "")
Meta$subtype <- str_replace_all(Meta$subtype," ", "_")


##double-check that entries are distinct
#duplicated PID from primary samples (normal samples often have a match)
#different tumors from the same patient for B1997.28921 and B2000.35726
#two replicates for B1997.23862 (internal control?)
dup <- Meta %>% 
  filter(type == "primary") %>%
  count(PID) %>% 
  filter(n>1) %>% 
  pull(PID)

Meta_PID_dup <- Meta %>% filter(PID %in% dup) %>% arrange(PID)

#kable(Meta_PID_dup, align = "c")

Tidy up data for processing and display

#Filter data for primary tumors
Meta_1 <- Meta %>% filter(type == "primary") %>% select(PID, gender, age, subtype, pT_simple, ISUP, necrose, lymphocytes, survival_months, censor_dead_0_alive_1)

Meta_1$gender <- str_replace_all(Meta_1$gender,"w","f")
Meta_1$subtype <- str_replace_all(Meta_1$subtype," ", "_")
Meta_1$necrose <- str_replace(Meta_1$necrose, "\\?", "")
Meta_1$lymphocytes <- str_replace(Meta_1$lymphocytes, "\\?", "")
#Note: Converted ISUP grade 4? into 4 (Patient: B2013.54303)
Meta_1$ISUP <- str_replace(Meta_1$ISUP, "\\?", "")

Meta_1[Meta_1== ""] <- NA

#Age bin for patients (five groups)
Meta_1$age <- as.numeric(Meta_1$age)
Meta_1 <- Meta_1 %>% mutate(age_bin = ifelse(age <= 40, "27-40", 
                                   ifelse(40<age & age<=55, "41-55",
                                          ifelse(55<age & age<=70, "56-70",
                                                 ifelse(70<age & age<=85, "71-85",
                                                        ifelse(85<age & age<=90, "85-88",NA))))))
#Survival data availability
Meta_1 <- Meta_1 %>% mutate(survival_data = ifelse(!is.na(survival_months),TRUE,FALSE))

#Save filtered metadata as csv
write.csv(Meta_1, file = "/Users/lasmey/Desktop/ccRCC/Metadaten_RCC/Metadata_combined_RCC.csv", row.names = F)

#Show overview of updated and processed metadata-table

Print heatmap based overview

based on the ComplexHeatmap package and the implemented HeatmapAnnotation function

library(ComplexHeatmap)
library(foreach)
library(ggsci)
library(RColorBrewer)

Meta_2 <- Meta_1

#Convert empty columns and NAs for easier handling with ComplexHeatmap package
Meta_2[is.na(Meta_2)] <- "Unknown"

Meta_2 <- Meta_2 %>% arrange(subtype, desc(ISUP), desc(pT_simple))

#Color scheme
#colnames(Meta_2)
col <- list(PID = colorRampPalette(brewer.pal(9, "Set1"))(length(unique(Meta_2$PID))),
           gender = c("#7FC97F", "#BEAED4"),
           age_bin = c(colorRampPalette(brewer.pal(9, "Greens"))(length(unique(Meta_2$age_bin))-1),"black"),
           #type = c("#E41A1C", "#377EB8"),
           subtype = c(colorRampPalette(brewer.pal(12, "Paired"))(length(unique(Meta_2$subtype))-1),"black"),
           pT_simple = c(colorRampPalette(brewer.pal(9, "Blues"))(length(unique(Meta_2$pT_simple))-1),"black"),
           ISUP = c(colorRampPalette(brewer.pal(9, "Reds"))(length(unique(Meta_2$ISUP))-1),"black"),
           necrose = c(colorRampPalette(brewer.pal(9, "Purples"))(length(unique(Meta_2$necrose))-1),"black"),
           lymphocytes = c(colorRampPalette(brewer.pal(9, "Greens"))(length(unique(Meta_2$lymphocytes))-1),"black"),
           survival_data = c("black","grey")
)

col_list_1 <- foreach(i=1:9, .packages="foreach") %do% {
  color= col[[i]]
  names(color)= sort(unique(Meta_2[,names(col)[i]]))
  color
}

names(col_list_1) <- c(names(col))

#Heatmap Annotation
sample_anno <- HeatmapAnnotation(PID = Meta_2$PID,
                                gender = Meta_2$gender,
                                age_bin = Meta_2$age_bin,
                                #type = Meta_2$type,
                                subtype = Meta_2$subtype,
                                ISUP = Meta_2$ISUP,
                                pT_simple = Meta_2$pT_simple,
                                necrose = Meta_2$necrose,
                                lymphocytes = Meta_2$lymphocytes,
                                survival_data = Meta_2$survival_data,
                                col = col_list_1,
                                border = T,
                                na_col = "black",
                                gap = unit(5, "points"),
                                show_annotation_name = T,
                                show_legend = c(PID = FALSE)
                                #annotation_label = c("PID (n=367)","Gender","Subtype","pT","pN")
)%v%NULL

#Display plot
draw(sample_anno)
